package th.co.scb.ace.demo.recovery.architecture;

import java.util.List;

public record ModuleMap(
        String rootPackage,
        List<Module> modules,
        List<External> external,
        List<String> clientLibraries) {
    public ModuleMap {
        modules = List.copyOf(modules);
        external = List.copyOf(external);
        clientLibraries = List.copyOf(clientLibraries);
    }

    public record Module(
            String code,
            String name,
            List<String> published,
            List<String> clientPackages) {
        public Module {
            published = List.copyOf(published);
            clientPackages = List.copyOf(clientPackages);
        }

        public String packageName(String rootPackage) {
            return rootPackage + "." + name;
        }
    }

    public record External(String name, String rootPackage, List<String> published) {
        public External {
            published = List.copyOf(published);
        }
    }
}
