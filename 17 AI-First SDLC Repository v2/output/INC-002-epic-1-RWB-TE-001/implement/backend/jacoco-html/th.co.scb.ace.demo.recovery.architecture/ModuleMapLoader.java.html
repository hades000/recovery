package th.co.scb.ace.demo.recovery.architecture;

import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.io.Reader;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;
import java.util.regex.Pattern;
import javax.lang.model.SourceVersion;
import org.yaml.snakeyaml.LoaderOptions;
import org.yaml.snakeyaml.Yaml;
import org.yaml.snakeyaml.constructor.SafeConstructor;
import org.yaml.snakeyaml.error.YAMLException;

public final class ModuleMapLoader {
    private static final Pattern MODULE_CODE = Pattern.compile("^(P([1-9]|10)|S[1-9])$");
    private static final Pattern MODULE_NAME = Pattern.compile("^[a-z][a-z0-9]*$");
    private static final Pattern PACKAGE_PATH =
            Pattern.compile("^[a-z][a-z0-9]*(\\.[a-z][a-z0-9]*)*$");
    private static final Set<String> ROOT_FIELDS =
            Set.of("rootPackage", "modules", "external", "clientLibraries");
    private static final Set<String> MODULE_FIELDS =
            Set.of("code", "name", "published", "clientPackages");
    private static final Set<String> EXTERNAL_FIELDS =
            Set.of("name", "rootPackage", "published");

    private ModuleMapLoader() {}

    public static ModuleMap load(Path file) {
        String fileName = file.toString();
        if (!Files.isRegularFile(file)) {
            throw invalid(fileName, "file", "does not exist");
        }

        try (Reader reader = Files.newBufferedReader(file, StandardCharsets.UTF_8)) {
            return parse(reader, fileName);
        } catch (IOException exception) {
            throw invalid(fileName, "file", "could not be read", exception);
        }
    }

    public static ModuleMap loadResource(String resource) {
        ClassLoader classLoader = ModuleMapLoader.class.getClassLoader();
        InputStream input = classLoader.getResourceAsStream(resource);
        if (input == null) {
            throw invalid(resource, "file", "was not found");
        }

        try (input; Reader reader = new InputStreamReader(input, StandardCharsets.UTF_8)) {
            return parse(reader, resource);
        } catch (IOException exception) {
            throw invalid(resource, "file", "could not be read", exception);
        }
    }

    private static ModuleMap parse(Reader reader, String file) {
        LoaderOptions options = new LoaderOptions();
        options.setAllowDuplicateKeys(false);

        Object loaded;
        try {
            loaded = new Yaml(new SafeConstructor(options)).load(reader);
        } catch (YAMLException exception) {
            throw invalid(file, "yaml", "is invalid", exception);
        }

        Map<?, ?> document = requireMap(loaded, file, "root");
        rejectUnknownFields(document, ROOT_FIELDS, file, "");

        String rootPackage = requiredString(document, "rootPackage", file, "");
        requireJavaPackage(rootPackage, file, "rootPackage");

        List<?> rawModules = requiredList(document, "modules", file, "", true);
        final List<ModuleMap.Module> modules = parseModules(rawModules, file);

        List<?> rawExternal = requiredList(document, "external", file, "", true);
        List<ModuleMap.External> external = parseExternal(rawExternal, file);
        requireExternal(external, "common", file);
        requireExternal(external, "collection", file);

        List<String> clientLibraries =
                parseStringList(
                        requiredList(document, "clientLibraries", file, "", true),
                        file,
                        "clientLibraries",
                        false,
                        false);
        validateUnique(clientLibraries, file, "clientLibraries");
        for (int index = 0; index < clientLibraries.size(); index++) {
            requireJavaPackage(
                    clientLibraries.get(index), file, "clientLibraries[" + index + "]");
        }

        return new ModuleMap(rootPackage, modules, external, clientLibraries);
    }

    private static List<ModuleMap.Module> parseModules(List<?> rawModules, String file) {
        List<ModuleMap.Module> modules = new ArrayList<>();
        Set<String> codes = new HashSet<>();
        Set<String> names = new HashSet<>();

        for (int index = 0; index < rawModules.size(); index++) {
            String field = "modules[" + index + "]";
            Map<?, ?> rawModule = requireMap(rawModules.get(index), file, field);
            rejectUnknownFields(rawModule, MODULE_FIELDS, file, field);

            String code = requiredString(rawModule, "code", file, field);
            if (!MODULE_CODE.matcher(code).matches()) {
                throw invalid(file, field + ".code", "has an invalid value");
            }
            if (!codes.add(code)) {
                throw invalid(file, field + ".code", "is duplicated");
            }

            String name = requiredString(rawModule, "name", file, field);
            if (!MODULE_NAME.matcher(name).matches()) {
                throw invalid(file, field + ".name", "has an invalid value");
            }
            if (!names.add(name)) {
                throw invalid(file, field + ".name", "is duplicated");
            }

            List<String> published =
                    parseStringList(
                            requiredList(rawModule, "published", file, field, true),
                            file,
                            field + ".published",
                            true,
                            true);
            validateUnique(published, file, field + ".published");
            for (int packageIndex = 0; packageIndex < published.size(); packageIndex++) {
                String packageName = published.get(packageIndex);
                if (packageName.equals("internal") || packageName.startsWith("internal.")) {
                    throw invalid(
                            file,
                            field + ".published[" + packageIndex + "]",
                            "cannot expose internal packages");
                }
            }

            List<String> clientPackages =
                    parseStringList(
                            requiredList(rawModule, "clientPackages", file, field, false),
                            file,
                            field + ".clientPackages",
                            false,
                            true);
            validateUnique(clientPackages, file, field + ".clientPackages");

            modules.add(new ModuleMap.Module(code, name, published, clientPackages));
        }

        return modules;
    }

    private static List<ModuleMap.External> parseExternal(List<?> rawExternal, String file) {
        List<ModuleMap.External> external = new ArrayList<>();
        Set<String> names = new HashSet<>();

        for (int index = 0; index < rawExternal.size(); index++) {
            String field = "external[" + index + "]";
            Map<?, ?> rawEntry = requireMap(rawExternal.get(index), file, field);
            rejectUnknownFields(rawEntry, EXTERNAL_FIELDS, file, field);

            String name = requiredString(rawEntry, "name", file, field);
            if (!MODULE_NAME.matcher(name).matches()) {
                throw invalid(file, field + ".name", "has an invalid value");
            }
            if (!names.add(name)) {
                throw invalid(file, field + ".name", "is duplicated");
            }

            String rootPackage = requiredString(rawEntry, "rootPackage", file, field);
            requireJavaPackage(rootPackage, file, field + ".rootPackage");

            List<String> published =
                    parseStringList(
                            requiredList(rawEntry, "published", file, field, true),
                            file,
                            field + ".published",
                            true,
                            true);
            validateUnique(published, file, field + ".published");
            external.add(new ModuleMap.External(name, rootPackage, published));
        }

        return external;
    }

    private static List<String> parseStringList(
            List<?> values,
            String file,
            String field,
            boolean mustNotBeEmpty,
            boolean packagePaths) {
        if (mustNotBeEmpty && values.isEmpty()) {
            throw invalid(file, field, "must not be empty");
        }

        List<String> result = new ArrayList<>();
        for (int index = 0; index < values.size(); index++) {
            String itemField = field + "[" + index + "]";
            Object value = values.get(index);
            if (!(value instanceof String text) || text.isBlank()) {
                throw invalid(file, itemField, "must be a non-empty string");
            }
            if (packagePaths && !PACKAGE_PATH.matcher(text).matches()) {
                throw invalid(file, itemField, "has an invalid package path");
            }
            result.add(text);
        }
        return result;
    }

    private static void requireExternal(
            List<ModuleMap.External> external, String name, String file) {
        if (external.stream().noneMatch(entry -> entry.name().equals(name))) {
            throw invalid(file, "external." + name, "is required");
        }
    }

    private static void validateUnique(List<String> values, String file, String field) {
        Set<String> unique = new HashSet<>();
        for (int index = 0; index < values.size(); index++) {
            if (!unique.add(values.get(index))) {
                throw invalid(file, field + "[" + index + "]", "is duplicated");
            }
        }
    }

    private static void requireJavaPackage(String value, String file, String field) {
        if (!SourceVersion.isName(value)) {
            throw invalid(file, field, "is not a valid Java name");
        }
    }

    private static Map<?, ?> requireMap(Object value, String file, String field) {
        if (!(value instanceof Map<?, ?> map)) {
            throw invalid(file, field, "must be a mapping");
        }
        return map;
    }

    private static List<?> requiredList(
            Map<?, ?> values, String key, String file, String parent, boolean mustNotBeEmpty) {
        String field = parent.isEmpty() ? key : parent + "." + key;
        Object value = values.get(key);
        if (!(value instanceof List<?> list)) {
            throw invalid(file, field, "must be a list");
        }
        if (mustNotBeEmpty && list.isEmpty()) {
            throw invalid(file, field, "must not be empty");
        }
        return list;
    }

    private static String requiredString(
            Map<?, ?> values, String key, String file, String parent) {
        String field = parent.isEmpty() ? key : parent + "." + key;
        Object value = values.get(key);
        if (!(value instanceof String text) || text.isBlank()) {
            throw invalid(file, field, "must be a non-empty string");
        }
        return text;
    }

    private static void rejectUnknownFields(
            Map<?, ?> values, Set<String> allowedFields, String file, String parent) {
        for (Object key : values.keySet()) {
            String field = key instanceof String text ? text : String.valueOf(key);
            if (!(key instanceof String) || !allowedFields.contains(field)) {
                throw invalid(file, parent.isEmpty() ? field : parent + "." + field, "is unknown");
            }
        }
    }

    private static IllegalArgumentException invalid(String file, String field, String detail) {
        return new IllegalArgumentException(file + ": " + field + " " + detail);
    }

    private static IllegalArgumentException invalid(
            String file, String field, String detail, Exception cause) {
        return new IllegalArgumentException(file + ": " + field + " " + detail, cause);
    }
}
