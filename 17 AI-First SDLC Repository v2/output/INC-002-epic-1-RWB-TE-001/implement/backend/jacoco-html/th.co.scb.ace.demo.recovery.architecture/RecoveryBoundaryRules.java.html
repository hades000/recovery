package th.co.scb.ace.demo.recovery.architecture;

import static com.tngtech.archunit.lang.syntax.ArchRuleDefinition.classes;

import com.tngtech.archunit.core.domain.Dependency;
import com.tngtech.archunit.core.domain.JavaClass;
import com.tngtech.archunit.core.domain.JavaClasses;
import com.tngtech.archunit.core.domain.JavaModifier;
import com.tngtech.archunit.lang.ArchCondition;
import com.tngtech.archunit.lang.ArchRule;
import com.tngtech.archunit.lang.ConditionEvents;
import com.tngtech.archunit.lang.SimpleConditionEvent;
import jakarta.persistence.Entity;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;
import org.springframework.data.repository.Repository;
import org.springframework.modulith.NamedInterface;

final class RecoveryBoundaryRules {
    private final ModuleMap moduleMap;

    private RecoveryBoundaryRules(ModuleMap moduleMap) {
        this.moduleMap = moduleMap;
    }

    static RecoveryBoundaryRules from(ModuleMap moduleMap) {
        return new RecoveryBoundaryRules(moduleMap);
    }

    ArchRule bnd1() {
        return externalDependencyRule("BND-1", "collection");
    }

    ArchRule bnd2() {
        return externalDependencyRule("BND-2", "common");
    }

    ArchRule bnd4() {
        return dependencyRule(
                "BND-4",
                "entity or repository",
                (source, target) ->
                        isEntityOrRepository(target)
                                && !belongsToSameRecoveryModule(source, target),
                targetModuleLabel("common"));
    }

    ArchRule bnd5() {
        return dependencyRule(
                "BND-5",
                "client library",
                (source, target) ->
                        matchesClientLibrary(target.getName())
                                && !isClientPackage(
                                        moduleFor(source.getPackageName()),
                                        source.getPackageName()),
                ignored -> clientPackageLabel());
    }

    List<String> moduleMapViolations(JavaClasses javaClasses) {
        List<String> violations = new ArrayList<>();
        Set<String> presentModules = new HashSet<>();
        Set<String> unmappedModules = new HashSet<>();
        Set<String> namedInterfacePackages = new HashSet<>();

        for (JavaClass javaClass : javaClasses) {
            String packageName = javaClass.getPackageName();
            if (!isInOrUnder(packageName, moduleMap.rootPackage())) {
                continue;
            }

            if (packageName.equals(moduleMap.rootPackage())
                    && !javaClass.getSimpleName().equals("package-info")) {
                violations.add(
                        "type "
                                + javaClass.getName()
                                + " sits in the root package "
                                + moduleMap.rootPackage());
            }
            if (javaClass.getSimpleName().equals("package-info")
                    && javaClass.isAnnotatedWith(NamedInterface.class)) {
                namedInterfacePackages.add(packageName);
            } else if (javaClass.isAnnotatedWith(NamedInterface.class)) {
                violations.add(
                        "type-level @NamedInterface on "
                                + javaClass.getName()
                                + "; declare published packages in package-info"
                                + " and the module map");
            }

            if (!packageName.equals(moduleMap.rootPackage())) {
                String firstSegment = packageName.substring(moduleMap.rootPackage().length() + 1);
                int separator = firstSegment.indexOf('.');
                String moduleName =
                        separator < 0 ? firstSegment : firstSegment.substring(0, separator);
                String modulePackage = moduleMap.rootPackage() + "." + moduleName;
                boolean moduleIsMapped =
                        moduleMap.modules().stream()
                                .anyMatch(module -> module.name().equals(moduleName));
                if (moduleIsMapped) {
                    presentModules.add(moduleName);
                    ModuleMap.Module module = moduleNamed(moduleName);
                    if (packageName.equals(modulePackage)
                            && !javaClass.getSimpleName().equals("package-info")
                            && javaClass.getModifiers().contains(JavaModifier.PUBLIC)) {
                        violations.add(
                                "public type "
                                        + javaClass.getName()
                                        + " in module base package "
                                        + modulePackage
                                        + "; move it to a published package or internal");
                    }
                } else {
                    unmappedModules.add(modulePackage);
                }
            }
        }

        for (String unmappedModule : unmappedModules) {
            violations.add("package " + unmappedModule + " is not in the module map");
        }

        for (ModuleMap.Module module : moduleMap.modules()) {
            String modulePackage = module.packageName(moduleMap.rootPackage());
            if (!presentModules.contains(module.name())) {
                violations.add("module package " + modulePackage + " does not exist");
            }

            for (String namedPackage : namedInterfacePackages) {
                if (isInOrUnder(namedPackage, modulePackage)) {
                    String relativePackage = namedPackage.substring(modulePackage.length() + 1);
                    if (!module.published().contains(relativePackage)) {
                        violations.add(
                                "@NamedInterface package "
                                        + namedPackage
                                        + " is not listed in published");
                    }
                }
            }

            for (String published : module.published()) {
                String publishedPackage = modulePackage + "." + published;
                boolean packageExists =
                        javaClasses.stream()
                                .anyMatch(
                                        javaClass ->
                                                isInOrUnder(
                                                        javaClass.getPackageName(),
                                                        publishedPackage));
                if (packageExists && !namedInterfacePackages.contains(publishedPackage)) {
                    violations.add(
                            "published package "
                                    + publishedPackage
                                    + " is missing @NamedInterface");
                }
            }
        }

        return List.copyOf(violations);
    }

    private ArchRule externalDependencyRule(String ruleCode, String externalName) {
        ModuleMap.External external = externalNamed(externalName);
        return dependencyRule(
                ruleCode,
                externalName,
                (source, target) ->
                        isInOrUnder(target.getPackageName(), external.rootPackage())
                                && external.published().stream()
                                        .noneMatch(
                                                published ->
                                                        isInOrUnder(
                                                                target.getPackageName(),
                                                                external.rootPackage()
                                                                        + "."
                                                                        + published)),
                ignored -> externalName);
    }

    private ArchRule dependencyRule(
            String ruleCode,
            String description,
            DependencyPredicate predicate,
            ModuleLabel moduleLabel) {
        return classes()
                .that()
                .resideInAPackage(moduleMap.rootPackage() + "..")
                .should(
                        new ArchCondition<>("depend on " + description) {
                            @Override
                            public void check(JavaClass source, ConditionEvents events) {
                                for (Dependency dependency :
                                        source.getDirectDependenciesFromSelf()) {
                                    JavaClass target = dependency.getTargetClass();
                                    if (predicate.matches(source, target)) {
                                        String recommendation =
                                                ruleCode.equals("BND-5")
                                                        ? "Use "
                                                                + moduleLabel.label(target)
                                                                + " or an application event."
                                                        : "Use "
                                                                + moduleLabel.label(target)
                                                                + ".api or an application event.";
                                        String message =
                                                "Rule "
                                                        + ruleCode
                                                        + " broken: "
                                                        + source.getName()
                                                        + " depends on "
                                                        + target.getName()
                                                        + ". "
                                                        + recommendation;
                                        events.add(SimpleConditionEvent.violated(source, message));
                                    }
                                }
                            }
                        })
                .allowEmptyShould(true);
    }

    private boolean isEntityOrRepository(JavaClass target) {
        return target.isAnnotatedWith(Entity.class) || target.isAssignableTo(Repository.class);
    }

    private String moduleFor(String packageName) {
        if (!isInOrUnder(packageName, moduleMap.rootPackage())
                || packageName.equals(moduleMap.rootPackage())) {
            return null;
        }
        String relativePackage = packageName.substring(moduleMap.rootPackage().length() + 1);
        int separator = relativePackage.indexOf('.');
        String moduleName =
                separator < 0 ? relativePackage : relativePackage.substring(0, separator);
        return moduleMap.modules().stream()
                .filter(module -> module.name().equals(moduleName))
                .map(ModuleMap.Module::name)
                .findFirst()
                .orElse(null);
    }

    private boolean belongsToSameRecoveryModule(JavaClass source, JavaClass target) {
        String sourceModule = moduleFor(source.getPackageName());
        return sourceModule != null && sourceModule.equals(moduleFor(target.getPackageName()));
    }

    private ModuleLabel targetModuleLabel(String fallback) {
        return target -> {
            String targetModule = moduleFor(target.getPackageName());
            if (targetModule != null) {
                return targetModule;
            }
            return moduleMap.external().stream()
                    .filter(
                            external ->
                                    isInOrUnder(
                                            target.getPackageName(), external.rootPackage()))
                    .map(ModuleMap.External::name)
                    .findFirst()
                    .orElse(fallback);
        };
    }

    private boolean matchesClientLibrary(String className) {
        return moduleMap.clientLibraries().stream()
                .anyMatch(
                        entry ->
                                className.equals(entry)
                                        || className.startsWith(entry + ".")
                                        || className.startsWith(entry + "$"));
    }

    private boolean isClientPackage(String moduleName, String packageName) {
        return moduleMap.modules().stream()
                .filter(module -> module.name().equals(moduleName))
                .anyMatch(
                        module ->
                                module.clientPackages().stream()
                                        .map(
                                                clientPackage ->
                                                        module.packageName(moduleMap.rootPackage())
                                                                + "."
                                                                + clientPackage)
                                        .anyMatch(
                                                clientPackage ->
                                                        isInOrUnder(packageName, clientPackage)));
    }

    private String clientPackageLabel() {
        return moduleMap.modules().stream()
                .filter(module -> !module.clientPackages().isEmpty())
                .map(
                        module ->
                                module.name()
                                        + "."
                                        + module.clientPackages().getFirst())
                .reduce((first, second) -> first + " or " + second)
                .orElse("an approved client package");
    }

    private ModuleMap.Module moduleNamed(String name) {
        return moduleMap.modules().stream()
                .filter(module -> module.name().equals(name))
                .findFirst()
                .orElseThrow();
    }

    private ModuleMap.External externalNamed(String name) {
        return moduleMap.external().stream()
                .filter(external -> external.name().equals(name))
                .findFirst()
                .orElseThrow();
    }

    private static boolean isInOrUnder(String packageName, String packageRoot) {
        return packageName.equals(packageRoot) || packageName.startsWith(packageRoot + ".");
    }

    @FunctionalInterface
    private interface DependencyPredicate {
        boolean matches(JavaClass source, JavaClass target);
    }

    @FunctionalInterface
    private interface ModuleLabel {
        String label(JavaClass target);
    }
}
